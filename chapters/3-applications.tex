\section{Frobenius Norm Approximation}
\label{sec:frobenius-norm-approximation}

The acceptance probability of Algorithm~\ref{alg:sampling-procedure} gives an immediate estimator for $\|\bA\|_F^2$. This section records the estimator separately because it will be used again in the eigenvalue approximation application.

\begin{corollary}[Frobenius norm approximation]
    Fix $\epsilon, \delta > 0$. Suppose we run $r$ rounds of Algorithm~\ref{alg:sampling-procedure}. Let $X$ be the number of accepted rounds and compute 
    \[\widehat F := \frac{X}{r}\tr(\bA)^2.\]
    If $r\geq 3n\log(2/\delta)/\epsilon^2$ then with probability at least $1-\delta$,
    \[(1-\epsilon)\|\bA\|^2_F\leq \widehat F\leq(1+\epsilon)\|\bA\|^2_F.\]
\end{corollary}

\begin{proof}
    Define the random variable
    \[ X_i :=\begin{cases}1&\text{if round $i$ accepted}\\
    0&\text{otherwise.}\end{cases}\]
    Recall from the proof of Theorem~\ref{thm:fast-l2-sampling} that the acceptance probability for any round is
    \[\beta = \frac{\|\bA\|^2_F}{\tr(\bA)^2}\]
    so $\E X_i = \beta$.
\end{proof}

\section{Robust Low-Rank Approximation}
Consider a matrix of the form $\bB=\bA+\bN\in\R^{n\times n}$, where $\bA$ is an unknown positive semidefinite matrix and $\bN$ is an unknown corruption matrix. This setting models the noise that may occur in real-world computation, for example, due to rounding or numerical errors. With some mild assumptions on $\bN$ and only query access to the matrix $\bB$, can we still hope to recover a good low-rank approximation to $\bA$?


Bakshi, Chepurko, and Woodruff \cite{bakshi_robust_2020} study low-rank approximation in this setting under the assumptions
\[
    \norm{\bN}_F^2\leq \eta\norm{\bA}_F^2
    \qquad\text{and}\qquad
    \|\bN_{i,*}\|_2^2\leq c\|\bA_{i,*}\|_2^2
    \quad\text{for all }i\in[n],
\]
for a fixed constant $c$. In words, the Frobenius norm of the corruption is bounded by that of $\bA$, and the corruption is well-spread among the rows. Their algorithm also assumes access to the parameter
\[
    \phi_{\max}:=\max_{i\in[n]}\frac{\bA_{ii}}{|\bB_{ii}|}.
\]
and achieves a rank-$k$ approximation $\bX$ satisfying
\[
    \norm{\bA-\bX}_F^2
    \leq
    \norm{\bA-\bA_k}_F^2+(\varepsilon+\sqrt{\eta})\norm{\bA}_F^2
\]
using $\widetilde O(\phi_{\max}^2nk/\varepsilon)$ queries to $\bB$. We present a simpler algorithm and analysis via fast length-squared sampling. The main idea is to modify our techniques to efficiently perform length-squared sampling on a matrix $\widetilde \bB$ derived from $\bB$. We then apply Algorithm~\ref{todo} to find an additive-error low rank approximation to $\widetilde \bB$. Finally, we argue that this approximation is also a good approximation to $\bA$.


Assume $\phi_{\max}$ is known, and let $\Phi = \max\{1, \phi_{\max}\}$. We first define the matrix $\widetilde{\bB}$ entrywise by
\[
    \widetilde{\bB}_{ij}
    :=
    \clip\!\left(
        \bB_{ij},
        -\Phi\sqrt{|\bB_{ii}|\,|\bB_{jj}|},
        \Phi\sqrt{|\bB_{ii}|\,|\bB_{jj}|}
    \right),
\]
where
\[
    \clip(x,-\tau,\tau):=\min\{\tau,\max\{-\tau,x\}\}.
\]
A query to $\widetilde{\bB}_{ij}$ is very efficient when the diagonal entries of $\bB$ are known.

\begin{lemma}[Clipping lemma]
The matrix $\widetilde\bB$ satisfies
\[
    \widetilde{\bB}_{ij}^{\,2}
    \leq
    \Phi^2|\bB_{ii}|\,|\bB_{jj}|
    \qquad\text{for all }i,j\in[n].
\]
Moreover, we may write
\[\widetilde\bB = \bA + \widetilde\bN\]
for some matrix $\widetilde\bN$ satisfying $\|\widetilde\bN\|^2_F\leq \|\bN\|^2_F$. 
\end{lemma}

\begin{proof}
The first claim follows directly from the definition of clipping. For the
second claim, we first note
\[
    \bA_{ii}\leq \Phi|\bB_{ii}|
    \qquad\text{and}\qquad
    \bA_{jj}\leq \Phi|\bB_{jj}|.
\]
by the definition of $\phi_{\max}$. Moreover, $\bA_{ij}^2\leq \bA_{ii}\bA_{jj}$ since $\bA$ is psd, so
\[
    |\bA_{ij}|
    \leq
    \Phi\sqrt{|\bB_{ii}|\,|\bB_{jj}|}.
\]
So the clipping interval for entry $ij$ contains 
$\bA_{ij}$. In particular,
\[
    |\bB_{ij}-\bA_{ij}|
    =
    |\bB_{ij}-\widetilde{\bB}_{ij}|
    +
    |\widetilde{\bB}_{ij}-\bA_{ij}|
    \geq
    |\widetilde{\bB}_{ij}-\bA_{ij}|.
\]
Therefore
\[
    (\widetilde{\bB}_{ij}-\bA_{ij})^2
    \leq
    \left(\bB_{ij}-\bA_{ij}\right)^2
    =
    \bN_{ij}^2.
\]
Summing over all entries gives $\|\widetilde\bB - \bA\|^2_F\leq \|\bN\|^2_F$, which is the desired result.
\end{proof}

Given entry access to $\bB$, we now demonstrate how to obtain a length-squared sample from $\widetilde\bB$. Let
\[
    D:=\sum_{i=1}^n |\bB_{ii}|.
\]
Assume $D>0$. In one trial, 
\begin{enumerate}
    \item Sample indices $i,j$ independently from $\frac{\diag(|\bB|)}{D}$.
    \item Accept $i$ with probability
    $\frac{\widetilde{\bB}_{ij}^{\,2}}{\Phi^2|\bB_{ii}|\,|\bB_{jj}|}.$
\end{enumerate}

\begin{lemma}
Conditioned on acceptance,
\[
    p_i
    =
    \frac{\|\widetilde{\bB}_{i,*}\|_2^2}{\|\widetilde{\bB}\|_F^2}
\]
and the expected number of trials until acceptance is at most
$\Phi^2 n$.
\end{lemma}

\begin{theorem}[Robust low-rank approximation.]
    Let $\bM$ be the approximation to $\tilde\bB$ obtained via \cite{frieze_fast_2004}. Then $\bM$ satisfies
    \[\|\bA - \bM\|^2_F\leq \|\bA - \bA_k\|^2_F + O(\epsilon + \sqrt\eta)\|\bA\|^2_F.\]
    Moreover, computing $\bM$ requires only $O(\phi_{\max}^2nk/\epsilon)$ entries of $\bB$.
\end{theorem}

\begin{proof}
    By the guarantee of \cite{frieze_fast_2004},
    \[\|\tilde \bB - \bM\|^2_F\leq \|\tilde B - \tilde B_k\|^2_F + \epsilon \|\tilde B\|^2_F.\]
    Expanding terms,
    \begin{align*}
        \|\bA + \tilde \bN - \bM\|^2_F&\leq \|\bA + \tilde \bN - (\bA + \tilde \bN)_k\|^2_F + \epsilon \|\bA + \tilde \bN\|^2_F\\
        &\leq \|\bA + \tilde \bN - \bA_k\|^2_F + \epsilon \|\bA + \tilde \bN\|^2_F\\
        &\leq \|\bA - \bA_k\|^2_F + \|\tilde\bN\|^2_F + 2\langle\bA - \bA_k, \tilde\bN\rangle + (3 + \eta)\epsilon\|\bA\|^2_F\\
        &\leq \|\bA - \bA_k\|^2_F + O(\epsilon + \sqrt\eta)\|\bA\|^2_F.
    \end{align*}
    In addition,
    \begin{align*}
        \|\bA + \tilde \bN - \bM\|^2_F &= \|\bA - \bM\|^2_F + \|\tilde\bN\|^2_F + 2\langle\bA - \bM, \tilde\bN\rangle\\
        &\geq\|\bA - \bM\|^2_F - 2\sqrt\eta\|\bA\|^2_F.
    \end{align*}
    Putting these together gives 
    \[\|\bA - \bM\|^2_F\leq \|\bA - \bA_k\|^2_F + O(\epsilon + \sqrt\eta)\|\bA\|^2_F.\]
    as desired.
\end{proof}
