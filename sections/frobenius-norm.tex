In this section, we extend our techniques to estimate the Frobenius norm of a psd matrix. Recall that in Algorithm~\ref{alg:sampling-procedure} of Chapter~\ref{chapter:fast-sample}, we proceed in rounds of adaptive sampling to obtain a length-squared sample. In the proof of Theorem~\ref{thm:fast-l2-sampling}, we show that the probability of acceptance for each round is $\|\bA\|^2_F/\tr(\bA)^2$. Thus, to estimate $\|\bA\|^2_F$, we simply estimate the acceptance probability and multiply by $\tr(\bA)^2$. 

In Theorem~\ref{thm:frobenius-norm-approximation-algorithm}, we will show that using only $O(n/\epsilon^2)$ entries of $\bA$, one may compute an estimate $\widehat F$ that is within $(1\pm \epsilon)$ of $\|\bA\|^2_F$ with constant probability. If $1/\epsilon^2$ is small enough compared to $n$, our algorithm does not need to read all entries of $\bA$. In Section~\ref{sec:eigenvalue-apporximation}, we apply this result to give a sublinear algorithm for approximating the full spectrum of a psd matrix.

We begin with the following calculation to determine the number of rejection sampling rounds required to obtain an accurate estimate of the acceptance probability, and hence $\|\bA\|^2_F$.

\begin{lemma}
\label{lem:frobenius-norm-approximation}
    Fix $\epsilon,\delta\in(0, 1)$. Suppose we run $r$ rounds of Algorithm~\ref{alg:sampling-procedure}. Let $X$ be the number of accepted rounds and compute 
    \[\widehat F := \frac{X}{r}\tr(\bA)^2.\]
    If $r\geq 3n\log(2/\delta)/\epsilon^2$ then with probability at least $1-\delta$,
    \[(1-\epsilon)\|\bA\|^2_F\leq \widehat F\leq(1+\epsilon)\|\bA\|^2_F.\]
\end{lemma}

\begin{proof}
    Let $X_i$ be an indicator random variable where $X_i = 1$ if round $i$ is accepted and $X_i=0$ otherwise. Recall from the proof of Theorem~\ref{thm:fast-l2-sampling} that the acceptance probability for any round is
    \[\beta = \frac{\|\bA\|^2_F}{\tr(\bA)^2}\]
    so $\E X_i = \beta$. Since the rounds are independent,
    \(
        X=\sum_{i=1}^r X_i
    \)
    is a sum of independent indicator random variables with $\E X = r\beta.$
    By the Chernoff bound,
    \[
        \Pr\left[|X-r\beta|\geq \epsilon r\beta\right]
        \leq 2\exp\left(-\frac{\epsilon^2 r\beta}{3}\right).
    \]
    From the proof of Theorem~\ref{thm:fast-l2-sampling}, we also have
    $\beta\geq 1/n$. Therefore, if
    \(
        r\geq {3n\log(2/\delta)}/{\epsilon^2},
    \)
    then
    \[
        2\exp\left(-\frac{\epsilon^2 r\beta}{3}\right)
        \leq
        2\exp\left(-\frac{\epsilon^2 r}{3n}\right)
        \leq \delta.
    \]
    Thus, with probability at least $1-\delta$,
    \[
        (1-\epsilon)\beta
        \leq
        \frac{X}{r}
        \leq
        (1+\epsilon)\beta.
    \]
    Multiplying through by $\tr(\bA)^2$ gives the desired result.
\end{proof}

\begin{algorithm}[H]
\caption{Frobenius Norm Approximation}
\label{alg:frobenius-norm-approximation}

\noindent
Let \(d\) be the diagonal distribution with $d_i = \bA_{ii}/\tr(\bA)$.

\smallskip
Set \(r=\left\lceil 3n\log(200)/\epsilon^2\right\rceil\). Run the following procedure for \(r\) independent rounds:
\begin{enumerate}[
    label=\arabic*.,
    leftmargin=2em,
    itemsep=0.35em,
    topsep=0.45em,
    parsep=0pt
]
    \item Sample indices \(i,j\overset{\mathrm{i.i.d.}}{\sim}d\).

    \item Accept with probability
    \(
        \frac{\bA_{ij}^2}{\bA_{ii}\bA_{jj}}.
    \)
\end{enumerate}

\smallskip
\noindent
Let \(X\) be the number of accepted rounds. Return
\[
    \widehat F:=\frac{X}{r}\tr(\bA)^2.
\]
\end{algorithm}

\begin{theorem}
\label{thm:frobenius-norm-approximation-algorithm}
Let \(\bA\in\R^{n\times n}\) be psd, and fix \(\epsilon\in(0,1)\). With probability at least 0.99, the estimate \(\widehat F\) returned by Algorithm~\ref{alg:frobenius-norm-approximation} satisfies
\[
    (1-\epsilon)\|\bA\|_F^2
    \leq
    \widehat F
    \leq
    (1+\epsilon)\|\bA\|_F^2.
\]
Moreover, the algorithm reads only \(O(n/\epsilon^2)\) entries of \(\bA\).
\end{theorem}

\begin{proof}
The error guarantee on $\widehat F$ follows from Lemma~\ref{lem:frobenius-norm-approximation} applied to $\delta  = 0.01$. The query complexity is also immediate from the choice of $r$.
\end{proof}
